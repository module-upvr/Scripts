local function CreatePart(name, parent, weld, offset) 
  local Part = Instance.new("Part")
  Part.Name = name
  Part.Massless = true
  Part.Anchored = false
  Part.CanCollide = false
  Part.CanTouch = false
  Part.CanQuery = false

  if weld then
    local Weld = Instance.new("Weld")
    Weld.C0 = offset or CFrame.new(0, 0, 0) 
    Weld.Part0 = Part
    Weld.Part1 = weld
    Weld.Parent = Part
  end

  Part.Parent = parent
  return Part
end

local function Accs(AccsName, parent, Mesh, Texture, Scale, CF, Welded, Angle) 
  Scale = Scale or Vector3.new(1, 1, 1)
  Angle = Angle or CFrame.Angles(0, 0, 0)
  CF = CF or CFrame.new(0, 0, 0)
  
  if typeof(Welded) == "string" then
    Welded = parent[Welded]
  end

  local var_2 = CreatePart(AccsName, parent) 
  var_2.Size = Vector3.new(1, 1, 1)

  local var_3 = Instance.new("SpecialMesh") 
  var_3.Name = "Mesh"
  var_3.MeshId = Mesh
  var_3.TextureId = Texture
  var_3.Scale = Scale
  var_3.Parent = var_2

  local var_4 = Instance.new("Weld") 
  var_4.C0 = CF * Angle
  var_4.Part0 = var_2
  var_4.Part1 = Welded
  var_4.Parent = var_2
  
  return var_2
end

local function Cloth(Id1, Id2, Id3, Parent)
  if Id1 ~= nil then
    local S = Instance.new("Shirt", Parent)
    S.ShirtTemplate = Id1
  end
  if Id2 ~= nil then
    local P = Instance.new("Pants", Parent) 
    P.PantsTemplate = Id2
  end
  if Id3 ~= nil then
    local SG = Instance.new("ShirtGraphic", Parent)
    SG.Graphic = Id3
  end
end

local function D(...)
  for _, v in ipairs({...}) do
    v:Destroy() 
  end
end

local function twotimedel(P)
  local tt = P:FindFirstChild("Life1") 
  local tt2 = P:FindFirstChild("Life2")
  if tt then
    for _, v in ipairs(tt:GetDescendants()) do
      if not table.find(ignoredel, v.Name) then
        v:Destroy()
      end
    end
  elseif tt2 then
    for _, v in ipairs(tt2:GetDescendants()) do
      if not table.find(ignoredel, v.Name) then
        v:Destroy()
      end
    end
  end
end

local function DoMorph(P, Name)

    local PosShirt = P:FindFirstChildOfClass("Shirt")
    local PosPants = P:FindFirstChildOfClass("Pants") 
    local PosCM = P:FindFirstChildOfClass("CharacterMesh")

    if PosPants then PosPants:Destroy() end
    if PosCM then
      if PosCM.BodyPart == Enum.BodyPart.Torso then
        PosCM:Destroy() 
      end
    end

    local CM = Instance.new("CharacterMesh", P) 
    CM.BodyPart = Enum.BodyPart.Torso
    CM.MeshId = 48112070

    madify(P)

    elseif P.Name == "Shedletsky" then

    elseif P.Name == "BuilderMan" then
