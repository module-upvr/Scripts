local function import(link, name)
  if not isfile(name) then
    writefile(name, game:HttpGet(link))
  end
  return getcustomasset(name)
end

local function CreatePart(name, parent, weld, offset) 
  local Part = Instance.new("Part")
  Part.Name = name
  Part.Massless = true
  Part.Anchored = false
  Part.CanCollide = false
  Part.CanTouch = false
  Part.CanQuery = false

  if weld then
    local Weld = Instance.new("Weld")
    Weld.C0 = offset or CFrame.new(0, 0, 0) 
    Weld.Part0 = Part
    Weld.Part1 = weld
    Weld.Parent = Part
  end

  Part.Parent = parent
  return Part
end

local function Accs(AccsName, parent, Mesh, Texture, Scale, CF, Welded, Angle) 
  Scale = Scale or Vector3.new(1, 1, 1)
  Angle = Angle or CFrame.Angles(0, 0, 0)
  CF = CF or CFrame.new(0, 0, 0)

  local var_2 = CreatePart(AccsName, parent) 
  var_2.Size = Vector3.new(1, 1, 1)

  local var_3 = Instance.new("SpecialMesh") 
  var_3.Name = "Mesh"
  var_3.MeshId = Mesh
  var_3.TextureId = Texture
  var_3.Scale = Scale
  var_3.Parent = var_2

  local var_4 = Instance.new("Weld") 
  var_4.C0 = CF * Angle
  var_4.Part0 = var_2
  var_4.Part1 = Welded
  var_4.Parent = var_2
  
  return var_2
end

local function Cloth(Id1, Id2, Id3, Parent)
  if Id1 ~= nil then
    local S = Instance.new("Shirt", Parent)
    S.ShirtTemplate = Id1
  end
  if Id2 ~= nil then
    local P = Instance.new("Pants", Parent) 
    P.PantsTemplate = Id2
  end
  if Id3 ~= nil then
    local SG = Instance.new("ShirtGraphic", Parent)
    SG.Graphic = Id3
  end
end

local function D(...)
  for _, v in ipairs({...}) do
    v:Destroy() 
  end
end

local Other = workspace.GameAssets.Teams.Other

local function DoMorph(P, Name)
  if P:GetAttribute("KillerName") == Name and P:GetAttribute("SkinName") == "Default" then
    clearC()

    P:WaitForChild("HumanoidRootPart") 
    P:WaitForChild("Accessories")
    D(P.Accessories, P.FaceBehavior) 
    
    local _, model = pcall(insertService.LoadLocalAsset, insertService, Rig)
    model.Parent = workspace
    model:MoveTo(P.HumanoidRootPart.Position + Vector3.new(0, 5, 0))
    workspace:WaitForChild(model.Name)
    
    local OWAccs = model.Accessories
    OWAccs.Name = "OWAccs"
    OWAccs:Clone().Parent = P
    local NewAccs = P.OWAccs
    NewAccs.Name = "NewAccessories"
    
    for _, v in ipairs(NewAccs:GetChildren()) do
      local BodyPart = v.Name
      if BodyPart == "RightArm" then
        BodyPart = "Right Arm"
      elseif BodyPart == "LeftArm" then
        BodyPart = "Left Arm"
      elseif BodyPart == "RightLeg" then
        BodyPart = "Right Leg"
      elseif BodyPart == "LeftLeg" then
        BodyPart = "Left Leg"
      elseif BodyPart == "HeadModel" then
        BodyPart = "Head"
      end
      for _, f in ipairs(v:GetDescendants()) do
        if f:IsA("WeldConstraint") and f.Part0 then
          f.Enabled = false
          local offset = f.Part0.CFrame:ToObjectSpace(f.Part1.CFrame)
          f.Part0 = P[BodyPart]
          f.Part1.CFrame = f.Part0.CFrame * offset
          f.Enabled = true
        elseif f:IsA("Weld") then
          f.Part1 = P[BodyPart]
        elseif f:IsA("BasePart") then
          f.Massless = true
          f.Anchored = false
          f.CanCollide = false
          f.CanTouch = false
          f.CanQuery = false
        end
      end
    end
    if P.Animations.ChaseTheme.IsPlaying == true then
      P.Animations.ChaseTheme:Stop() 
      P.Animations.ChaseTheme.SoundId = ChaseTheme
      P.Animations.ChaseTheme:Play() 
    else
      P.Animations.ChaseTheme.SoundId = ChaseTheme
    end
    for _, v in ipairs(model.HumanoidRootPart:GetChildren()) do
      if v:IsA("Sound") then
        if v.Name == "BoltStartup" then v.Name = "nil" end
        if v.Name == "Bolt" then v.Name = "BoltStartup" end
        if P.HumanoidRootPart:FindFirstChild(v.Name) then
          P.HumanoidRootPart[v.Name].SoundId = v.SoundId
        end
      end
    end
    model:Destroy() 
    local Screen = P.NewAccessories.Head.Face.SurfaceGui:FindFirstChild("Face") 
    C = LMSFrame.ChildAdded:Connect(function(v)
      if v.Name == "LMSIntro" then
        C:Disconnect()
        C = nil
        Screen.Image = TrollFace
        CurrentFace = TrollFace
        CurrentSize = UDim2.new(1, 0, 1, 0)
        CurrentPosition = UDim2.new(0, 0, 0, 0)
        Screen.Size = CurrentSize
        Screen.Position = CurrentPosition
        for _, f in ipairs(v.Killers:GetDescendants()) do
          if f.Name == "Sprite" and f.Image == "rbxassetid://136532644529539" then
            f.Image = LMSUI
          end
        end
      end
    end) 
    C2 = P.Humanoid.HealthChanged:Connect(function(v)
     if v <= P.Humanoid.MaxHealth / 2 then
        Screen.Image = LowFace
        Screen.Size = UDim2.new(1, 0, 1, 0)
        Screen.Position = UDim2.new(0, 0, 0, 0)
      else
        Screen.Image = CurrentFace
        Screen.Size = CurrentSize
        Screen.Position = CurrentPosition
      end
    end)
    C3 = Other.ChildAdded:Connect(function(v) 
      if v.Name == "Computer" and v:GetAttribute("Owner") == Player.Name then
        local ComPiv = v:GetPivot() 
        D(v:WaitForChild("Model"), v:WaitForChild("Screen")) 
        
        local NewCom = OWPC:Clone()
        NewCom.Parent = v
        NewCom.Name = "OWComputer"
        NewCom:PivotTo(ComPiv)
        
        NewCom.Model.PCModel.Union.Color = soulColors[math.random(1, 7)]
      end
    end) 
    P.Humanoid.Died:Connect(clearC)
  end 
end

DoMorph(CurrentRig, "Badware")
