local function CreatePart(name, parent, weld, offset) 
  local Part = Instance.new("Part")
  Part.Name = name
  Part.Massless = true
  Part.Anchored = false
  Part.CanCollide = false
  Part.CanTouch = false
  Part.CanQuery = false

  if weld then
    local Weld = Instance.new("Weld")
    Weld.C0 = offset or CFrame.new(0, 0, 0) 
    Weld.Part0 = Part
    Weld.Part1 = weld
    Weld.Parent = Part
  end

  Part.Parent = parent
  return Part
end

local function Accs(AccsName, parent, Mesh, Texture, Scale, CF, Welded, Angle) 
  Scale = Scale or Vector3.zero
  Angle = Angle or CFrame.identity
  CF = CF or CFrame.identity

  local var_2 = CreatePart(AccsName, parent) 
  var_2.Size = Vector3.new(1, 1, 1)

  local var_3 = Instance.new("SpecialMesh") 
  var_3.Name = "Mesh"
  var_3.MeshId = Mesh
  var_3.TextureId = Texture
  var_3.Scale = Scale
  var_3.Parent = var_2

  local var_4 = Instance.new("Weld") 
  var_4.C0 = CF * Angle
  var_4.Part0 = var_2
  var_4.Part1 = Welded
  var_4.Parent = var_2

  return var_2
end

local function Cloth(Id1, Id2, Id3, Parent)
  if Id1 ~= nil then
    local S = Instance.new("Shirt", Parent)
    S.ShirtTemplate = Id1
  end
  if Id2 ~= nil then
    local P = Instance.new("Pants", Parent) 
    P.PantsTemplate = Id2
  end
  if Id3 ~= nil then
    local SG = Instance.new("ShirtGraphic", Parent)
    SG.Graphic = Id3
  end
end

local function D(...)
  for _, v in ipairs({...}) do
    v:Destroy() 
  end
end

local C
local C2
local C3

local function clearC(v)
  if typeof(v) == "RBXScriptConnection" then
    v:Disconnect()
    v = nil
  else
    if C then C:Disconnect() C = nil end
    if C2 then C2:Disconnect() C2 = nil end
    if C3 then C3:Disconnect() C3 = nil end
  end
end

local mf = Color3.fromRGB(27, 42, 53)

local function DoMorph(P, Name, SN)
  if P.Name == Name and P:GetAttribute("SkinName") == SN then
    for _, v in ipairs({"Torso", "Right Leg", "Left Leg", "Right Arm", "Left Arm"}) do
      local L = P:WaitForChild(v)
      if L then
        local Suits = CreatePart("SuitPart", L, L)
        Suits.Size = L.Size
        Suits.Color = mf
        Suits.Material = Enum.Material.Air
        L.Transparency = 1
      end
    end
    D(P.Mafioso, P.Head:FindFirstChildOfClass("MeshPart"), P:FindFirstChild("Clothing"), P.Clothing, P.HumanoidRootPart["rbxassetid://98541221291764"])

    P.Head.Transparency = 0
    P.Head.Color = Color3.fromRGB(245, 205, 48)
    P.Head.face.Texture = "rbxasset://textures/face.png"

    local Fed = Accs("Fedora", P, "http://www.roblox.com/asset/?id=1029012", "", Vector3.new(1.213, 1.213, 1.213), CFrame.new(0, -0.675, 0.055), P.Head, nil)
    Fed.Mesh.TextureId = ""
    Fed.Color = mf

    C3 = P.HumanoidRootPart.ChildAdded:Connect(function(v)
      if v:IsA("Sound") then
        if v.SoundId == "rbxassetid://88325606095183" then
          v.SoundId = MCT
          v:Play()
        elseif v.Name == "VoicelineSFX" then
          if table.find(Kill, v.SoundId) then
            v.SoundId = ({Voicelines.Kill, Voicelines.Kill2, Voicelines.Kill3, Voicelines.Kill4})[math.random(4)]
            v:Play()
          elseif table.find(CN, v.SoundId) then
            v.SoundId = Voicelines.CN
            v:Play()
          elseif table.find(PD, v.SoundId) then
            v.SoundId = ({Voicelines.PD, Voicelines.PD2, Voicelines.PD3})[math.random(3)]
            v:Play()
          elseif table.find(WOH, v.SoundId) then
            v.SoundId = Voicelines.WOH
            v:Play()
          elseif table.find(WOM, v.SoundId) then
            v.SoundId = Voicelines.WOM
            v:Play()
          else
            v:Stop()
          end
        end
      end
    end)

    C = P.Humanoid.Died:Once(clearC)
    C2 = Player.CharacterAdded:Once(clearC)
  end
end

Player.CharacterAdded:Connect(function(Rig)
  DoMorph(Rig, "c00lkidd", "MafiosoC00l")
end)
