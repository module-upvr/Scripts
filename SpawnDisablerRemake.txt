local function getModule(Name, req) 
    local var

    for _, v in ipairs(Modules:GetDescendants()) do
        if v.Name == Name then
            var = v
        end
    end

    if var == nil then
        for _, v in ipairs(Sys:GetDescendants()) do
            if v.Name == Name then
                var = v
                break
            end
        end
    end

    if req then
        return require(var) 
    else
        return var
    end
end

local function D(...) 
    for _, v in pairs({...}) do
        if v then v:Destroy() end 
    end
end 

local function printTable(tbl, indent)
    indent = indent or ""
    for k, v in pairs(tbl) do
        if typeof(v) == "table" then
            print(indent .. tostring(k) .. ":")
            printTable(v, indent .. "  ") -- recurse with increased indentation
        else
            print(indent .. tostring(k) .. ": " .. tostring(v))
        end
    end
end

local function Accs(AccsName, char, Mesh, Texture, Scale, CF, Welded, Angle) 
    Scale = Scale or Vector3.new(1, 1, 1) 
    CF = CF or CFrame.new(0, 0, 0) 
    Angle = Angle or CFrame.Angles(0, 0, 0) 

    local var_1 = Instance.new("Accessory", char) 
    var_1.Name = AccsName
    local var_2 = Instance.new("Part", var_1) 
    var_2.Anchored = false
    var_2.Massless = true
    var_2.CanCollide = false
    var_2.Size = Vector3.new(1, 1, 1) 
    var_2.Name = "Handle"
    local var_3 = Instance.new("SpecialMesh", var_2) 
    var_3.Name = "Mesh"
    var_3.MeshId = Mesh
    var_3.TextureId = Texture
    var_3.Scale = Scale
    local var_4 = Instance.new("Weld", var_2) 
    var_4.C0 = CF * Angle
    var_4.Part0 = var_2
    var_4.Part1 = char[Welded]
    return var_4
end

loop = RunService.Heartbeat:Connect(function(dt) 
    if not HaloPart or not P:FindFirstChild("Head") then return end
    
    -- Tentukan kecepatan putar (misal: math.pi = setengah putaran per detik)
    rotation = rotation + dt * 1.3

    -- Buat CFrame posisi di atas kepala
    local baseTarget = P.Head.CFrame * CFrame.new(0, 0.6 + 0.317, 0)
    
    -- Gabungkan Posisi dengan Rotasi pada sumbu X
    -- Gunakan CFrame.Angles(X, Y, Z)

    HaloPart.CanCollide = false
    HaloPart.Anchored = true

    HaloPart.CFrame = HaloPart.CFrame:Lerp(finalTarget, 0.2)

            limb.Color = Color3.new(1, 1, 1)

    if not arg1 then return end

    if isEnraged ~= lastEnragedState then
        lastEnragedState = isEnraged -- Update status terakhir

            machete.Transparency = 1
        else
            task.delay(0.4, function() 
                if machete and not arg1.State.isEnraged then 
                    machete.Transparency = 0 
                end
            end)
        end
    end
end)
