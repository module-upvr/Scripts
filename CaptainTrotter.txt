local Player = game:GetService("Players").LocalPlayer

local CurrentRig = Player.Character

  module.Animations[i] = v

local function Accs(AccsName, char, Mesh, Texture, Scale, CF, Welded, Angle) 
  Scale = Scale or Vector3.new(1, 1, 1)
  Angle = Angle or CFrame.Angles(0, 0, 0)
  CF = CF or CFrame.new(0, 0, 0)
  if typeof(Welded) == "string" then
    Welded = char[Welded]
  end

  local var_1 = Instance.new("Accessory", char) 
  var_1.Name = AccsName
  local var_2 = Instance.new("Part", var_1) 
  var_2.Anchored = false
  var_2.Massless = true
  var_2.CanCollide = false
  var_2.Size = Vector3.new(1, 1, 1) 
  var_2.Name = "Handle"
  local var_3 = Instance.new("SpecialMesh", var_2) 
  var_3.Name = "Mesh"
  var_3.MeshId = Mesh
  var_3.TextureId = Texture
  var_3.Scale = Scale
  local var_4 = Instance.new("Weld", var_2) 
  var_4.C0 = CF * Angle
  var_4.Part0 = var_2
  var_4.Part1 = Welded
  return var_4
end

local function Cloth(Id1, Id2, Id3, Parent)
  if Id1 ~= nil then
    local S = Instance.new("Shirt", Parent)
    S.ShirtTemplate = Id1
  end
  if Id2 ~= nil then
    local P = Instance.new("Pants", Parent) 
    P.PantsTemplate = Id2
  end
  if Id3 ~= nil then
    local SG = Instance.new("ShirtGraphic", Parent)
    SG.Graphic = Id3
  end
end

local function D(...)
  for _, v in ipairs({...}) do
    v:Destroy()
  end
end

local function addMesh(part, id1, id2) 
  local T = Instance.new("SpecialMesh", part) 
  T.MeshType = Enum.MeshType.FileMesh
  T.MeshId = id1
  T.TextureId = id2
end

local function addFace(part, texture) 
  local decal = Instance.new("Decal", part) 
  decal.Texture = texture
end

local function CreatePart(name, parent, weld, offset) 
  local Part = Instance.new("Part", parent)
  Part.Name = name
  Part.Massless = true
  Part.Anchored = false
  Part.CanCollide = false
  Part.CanTouch = false
  Part.CanQuery = false
  if weld then
    local Weld = Instance.new("WeldConstraint", Part)
    Part.CFrame = offset and weld.CFrame * offset or weld.CFrame
    Weld.Part0 = Part
    Weld.Part1 = weld
  end
  return Part
end
