local function CreatePart(name, parent, weld, offset) 
  local Part = Instance.new("Part")
  Part.Name = name
  Part.Massless = true
  Part.Anchored = false
  Part.CanCollide = false
  Part.CanTouch = false
  Part.CanQuery = false

  if weld then
    local Weld = Instance.new("Weld")
    Weld.C0 = offset or CFrame.new(0, 0, 0) 
    Weld.Part0 = Part
    Weld.Part1 = weld
    Weld.Parent = Part
  end

  Part.Parent = parent
  return Part
end

local function Accs(AccsName, parent, Mesh, Texture, Scale, CF, Welded, Angle) 
  Scale = Scale or Vector3.new(1, 1, 1)
  Angle = Angle or CFrame.Angles(0, 0, 0)
  CF = CF or CFrame.new(0, 0, 0)
  
  if typeof(Welded) == "string" then
    Welded = parent[Welded]
  end

  local var_2 = CreatePart(AccsName, parent) 
  var_2.Size = Vector3.new(1, 1, 1)

  local var_3 = Instance.new("SpecialMesh") 
  var_3.Name = "Mesh"
  var_3.MeshId = Mesh
  var_3.TextureId = Texture
  var_3.Scale = Scale
  var_3.Parent = var_2

  local var_4 = Instance.new("Weld") 
  var_4.C0 = CF * Angle
  var_4.Part0 = var_2
  var_4.Part1 = Welded
  var_4.Parent = var_2
  
  return var_2
end

  if Id2 ~= nil then

  if Id3 ~= nil then

  end
end

local function stopAfterimage()
  if connection then
    connection:Disconnect()
    connection = nil
  end
end

local function createAfterimage(char)
  if not char or not char:FindFirstChildOfClass("Humanoid") then
    char = Player.Character or Player.Character:Wait()
  end
  local clone = char:Clone()
  clone.Humanoid.EvaluateStateMachine = false
  clone.ExtraModels.Helmet.Mesh.VertexColor = Vector3.new(0, 0, 0) 
  
  clone:PivotTo(clone:GetPivot() + (-clone:GetPivot().LookVector * 0.5))
  
  for _, v in ipairs(clone:GetDescendants()) do
    if v:IsA("Animation") or v:IsA("Folder") or v:IsA("Sound") then
      v:Destroy()
    elseif v:IsA("BasePart") and v.Transparency < 1 then
      v.Anchored = true
      v.CanCollide = false
      v.CanQuery = false
      v.CanTouch = false
      v.Transparency = 0.5
      local Back = v.CFrame + (-clone:GetPivot().LookVector * 4) 
      local tweenInfo = TweenInfo.new(0.5, Enum.EasingStyle.Linear, Enum.EasingDirection.Out)
      local tween = TweenService:Create(v, tweenInfo, {Transparency = 1, CFrame = Back})
      
      tween.Completed:Connect(function()
        clone:Destroy()
      end)
      tween:Play()
    elseif v:IsA("Highlight") then
      v.OutlineTransparency = 0.5
      local tweenInfo = TweenInfo.new(0.5, Enum.EasingStyle.Linear, Enum.EasingDirection.Out)
      local tween = TweenService:Create(v, tweenInfo, {OutlineTransparency = 1})
      
      tween:Play()
    end
  end
  clone.Parent = workspace.Misc
end

local function startAfterimage(interval, char)
  if connection then return end
  local lastTime = 0
  connection = RunService.Heartbeat:Connect(function()
    local currentTime = os.clock()
    if currentTime - lastTime >= (interval or 0.1) then
      lastTime = currentTime
      createAfterimage(char)
    end
  end)
end
