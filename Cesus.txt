local function Accs(AccsName, char, Mesh, Texture, Scale, CF, Welded, Angle) 
  Scale = Scale or Vector3.new(1, 1, 1)
  Angle = Angle or CFrame.Angles(0, 0, 0)
  CF = CF or CFrame.new(0, 0, 0)

  local var_1 = Instance.new("Accessory", char) 
  var_1.Name = AccsName
  local var_2 = Instance.new("Part", var_1) 
  var_2.Anchored = false
  var_2.Massless = true
  var_2.CanCollide = false
  var_2.Size = Vector3.new(1, 1, 1) 
  var_2.Name = "Handle"
  local var_3 = Instance.new("SpecialMesh", var_2) 
  var_3.Name = "Mesh"
  var_3.MeshId = Mesh
  var_3.TextureId = Texture
  var_3.Scale = Scale
  local var_4 = Instance.new("Weld", var_2) 
  var_4.C0 = CF * Angle
  var_4.Part0 = var_2
  var_4.Part1 = char[Welded]
  return var_2
end

local function Cloth(Id1, Id2, Id3, Parent)
  if Id1 ~= nil then
    local S = Instance.new("Shirt", Parent)
    S.ShirtTemplate = Id1
  end
  if Id2 ~= nil then
    local P = Instance.new("Pants", Parent) 
    P.PantsTemplate = Id2
  end
  if Id3 ~= nil then
    local SG = Instance.new("ShirtGraphic", Parent)
    SG.Graphic = Id3
  end
end

local function D(...)
  for _, v in ipairs({...}) do
    v:Destroy()
  end
end

local function FakeLimb(Limb) 
    local Part = Instance.new("Part", Limb) 
    Part.Name = "Fake"
    Part.Size = Limb.Size
    Part.CanCollide = false
    Part.Anchored = false
    Part.Massless = true
    local Weld = Instance.new("Weld", Part) 
    Weld.Part0 = Part
    Weld.Part1 = Limb
    return Part
end
