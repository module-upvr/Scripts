local function CreatePart(name, parent, weld, offset) 
  local Part = Instance.new("Part")
  Part.Name = name
  Part.Massless = true
  Part.Anchored = false
  Part.CanCollide = false
  Part.CanTouch = false
  Part.CanQuery = false

  if weld then
    local Weld = Instance.new("Weld")
    Weld.C0 = offset or CFrame.new(0, 0, 0) 
    Weld.Part0 = Part
    Weld.Part1 = weld
    Weld.Parent = Part
  end

  Part.Parent = parent
  return Part
end

local function Accs(AccsName, parent, Mesh, Texture, Scale, CF, Welded, Angle) 
  Scale = Scale or Vector3.new(1, 1, 1)
  Angle = Angle or CFrame.Angles(0, 0, 0)
  CF = CF or CFrame.new(0, 0, 0)

  local var_2 = CreatePart(AccsName, parent) 
  var_2.Size = Vector3.new(1, 1, 1)

  local var_3 = Instance.new("SpecialMesh") 
  var_3.Name = "Mesh"
  var_3.MeshId = Mesh
  var_3.TextureId = Texture
  var_3.Scale = Scale
  var_3.Parent = var_2

  local var_4 = Instance.new("Weld") 
  var_4.C0 = CF * Angle
  var_4.Part0 = var_2
  var_4.Part1 = Welded
  var_4.Parent = var_2
  
  return var_2
end

local function Cloth(Id1, Id2, Id3, Parent)
  if Id1 ~= nil then
    local S = Instance.new("Shirt", Parent)
    S.ShirtTemplate = Id1
  end
  if Id2 ~= nil then
    local P = Instance.new("Pants", Parent) 
    P.PantsTemplate = Id2
  end
  if Id3 ~= nil then
    local SG = Instance.new("ShirtGraphic", Parent)
    SG.Graphic = Id3
  end
end

local function D(...)
  for _, v in ipairs({...}) do
    v:Destroy() 
  end
end

local function CM(BodyPart, ID, parent)
  local mesh = Instance.new("CharacterMesh", parent)
  mesh.Name = string.format("%sMesh", BodyPart)
  mesh.MeshId = ID
  mesh.BodyPart = Enum.BodyPart[BodyPart]
end

local function DoMorph(P, Name)
  if P.Name == Name and P:GetAttribute("SkinName") == "" then
    for _, v in ipairs({"Head", "Torso", "Right Leg", "Left Leg", "Right Arm", "Left Arm"}) do
      local L = P:WaitForChild(v)
      if L then
        L.Transparency = 0
        L.Color = Color3.fromRGB(255,255,255)
        local NS = Vector3.new(math.round(L.Size.X / 1.121), math.round(L.Size.Y / 1.121), math.round(L.Size.Z / 1.121))
        L.Size = NS
      end
    end
    D(P.cool, P.HumanoidRootPart["rbxassetid://98541221291764"])

    P.Head.Mesh.MeshId = "rbxassetid://8635368147"
    P.Head.Mesh.Scale = Vector3.new(1, 1, 1)
    P.firebrand.Mesh.MeshId = "http://www.roblox.com/asset/?id=95891318"
    P.firebrand.Mesh.TextureId = "http://www.roblox.com/asset/?id=133863187"
    P.firebrand.Mesh.VertexColor = Vector3.new(1, 1, 1)
    P.firebrand.Mesh.Scale = Vector3.new(0.8, 0.8, 0.8)
    P["Left Arm"].firebrand.C0 = CFrame.new(0.015, -1.8, -5.5) * CFrame.Angles(-math.pi, -math.pi/2, -math.pi/2)
    P.Torso.Neck.C0 = CFrame.new(0, 0.95, 0) * CFrame.Angles(-math.pi/2, 0, -math.pi)
    
    C3 = P.Torso.Neck:GetPropertyChangedSignal("C0"):Connect(function(v)
      P.Torso.Neck.C0 = CFrame.new(0, 0.95, 0) * CFrame.Angles(-math.pi/2, 0, -math.pi)
    end)
    
    local Face = Instance.new("Decal", P.Head)
    Face.Texture = "rbxasset://textures/face.png"

    Cloth("http://www.roblox.com/asset/?id=9415501529", "http://www.roblox.com/asset/?id=144076759", nil, P)

    CM("Torso", "82907945", P)
    CM("LeftArm", "82907977", P)
    CM("RightArm", "82908019", P)
    CM("LeftLeg", "81487640", P)
    CM("RightLeg", "81487710", P)

    local Shades = Accs("Shades", P, "rbxassetid://4074491872", "http://www.roblox.com/asset/?id=4074505370", nil, CFrame.new(0, -0.325, 0.226), P.Head, nil)

    C4 = P.HumanoidRootPart.ChildAdded:Connect(function(v)
      if v:IsA("Sound") then
        if v.Name == "VoicelineSFX" then
          if table.find(KillTbl, v.SoundId) then
            local r = math.random(1, 4)
            v.SoundId = NewKillTbl[r]
            v:Play()
          elseif table.find(PizzaDeliveryTbl, v.SoundId) then
            v.SoundId = PD
            v:Play()
          elseif table.find(WalkspeedOverrideTbl, v.SoundId) then
            v.SoundId = WO
            v:Play()
          elseif table.find(WalkspeedOverrideCollideTbl, v.SoundId) then
            v.SoundId = WOCrash
            v:Play()
          elseif table.find(WalkspeedOverrideTimeoutTbl, v.SoundId) then
            v.SoundId = WOMiss
            v:Play()
          elseif table.find(StunnedTbl, v.SoundId) then
            local r = math.random(1, 2)
            v.SoundId = NewStunTbl[r]
            v:Play()
          else
            v:Stop()
            v:Destroy()
          end
        end
      end
    end)
    
    C5 = Themes.ChildAdded:Connect(function(v)
      if v:IsA("Sound") then
        if v.Name == "LastSurvivor" or v.SoundId == "rbxassetid://80564889711353" then
          v.SoundId = LMS
          v:Play()
        elseif v.Sound == "rbxassetid://72867661507332" then
          v.SoundId = ChaseTheme
          v:Play()
        end
      end
    end)
    
    C = P.Humanoid.Died:Connect(clearC)
    C2 = Player.CharacterAdded:Connect(clearC)
  end
end
