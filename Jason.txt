local function CreatePart(name, parent, weld, offset) 
  local Part = Instance.new("Part")
  Part.Name = name
  Part.Massless = true
  Part.Anchored = false
  Part.CanCollide = false
  Part.CanTouch = false
  Part.CanQuery = false

  if weld then
    local Weld = Instance.new("Weld")
    Weld.C0 = offset or CFrame.new(0, 0, 0) 
    Weld.Part0 = Part
    Weld.Part1 = weld
    Weld.Parent = Part
  end

  Part.Parent = parent
  return Part
end

local function Accs(AccsName, parent, Mesh, Texture, Scale, CF, Welded, Angle) 
  Scale = Scale or Vector3.new(1, 1, 1)
  Angle = Angle or CFrame.Angles(0, 0, 0)
  CF = CF or CFrame.new(0, 0, 0)

  local var_2 = CreatePart(AccsName, parent) 
  var_2.Size = Vector3.new(1, 1, 1)

  local var_3 = Instance.new("SpecialMesh") 
  var_3.Name = "Mesh"
  var_3.MeshId = Mesh
  var_3.TextureId = Texture
  var_3.Scale = Scale
  var_3.Parent = var_2

  local var_4 = Instance.new("Weld") 
  var_4.C0 = CF * Angle
  var_4.Part0 = var_2
  var_4.Part1 = Welded
  var_4.Parent = var_2
  
  return var_2
end

local function Cloth(Id1, Id2, Id3, Parent)
  if Id1 ~= nil then
    local S = Instance.new("Shirt", Parent)
    S.ShirtTemplate = Id1
  end
  if Id2 ~= nil then
    local P = Instance.new("Pants", Parent) 
    P.PantsTemplate = Id2
  end
  if Id3 ~= nil then
    local SG = Instance.new("ShirtGraphic", Parent)
    SG.Graphic = Id3
  end
end

local function D(...)
  for _, v in ipairs({...}) do
    v:Destroy() 
  end
end

local function DoMorph(P, Name)
  if P.Name == Name and P:GetAttribute("SkinName") == "" then
    for _, v in ipairs({"Head", "Torso", "Right Leg", "Left Leg", "Right Arm", "Left Arm"}) do
      local L = P:WaitForChild(v)
      if L then
        L.Transparency = 0
        L.BrickColor = BrickColor.new("Black")
        if L.Name == "Head" then
          L.BrickColor = BrickColor.new("Really black")
        end
      end
    end
    D(P["Slasher Export"], P.Chainsaw.Note, P.Mask)

    P.Machete.Transparency = 1
    P.Chainsaw.Transparency = 1
    
    Cloth("http://www.roblox.com/asset/?id=1502643168", "http://www.roblox.com/asset/?id=1502651303", nil, P)
    
    Accs("Mask", P, "http://www.roblox.com/asset/?id=5158270", "http://www.roblox.com/asset/?id=5158267", Vector3.new(0.13, 0.13, 0.1), CFrame.new(0, -0.6, 0), P.Head, CFrame.Angles(math.pi/2, 0, 0))

    Accs("OldMachete", P.Machete, "rbxassetid://441575918", "rbxassetid://441575955", Vector3.new(0.016, 0.016, 0.016), nil, P.Machete, CFrame.Angles(0, -math.pi/2, 0))

    Accs("OldChainsaw", P.Chainsaw, "http://www.roblox.com/asset/?id=2766469", "rbxasset://textures/SwordTexture.png", Vector3.new(0.89, 0.89, 1.157), nil, P.Chainsaw, CFrame.Angles(-math.pi/2, math.pi/2, 0))
  end
end

Player.CharacterAdded:Connect(function(Rig)
  DoMorph(Rig, "Slasher")
end)
