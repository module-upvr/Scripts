local Player = game:GetService("Players").LocalPlayer

local CurrentRig = Player.Character

local function import(link, name)
  if not isfile(name) then
    writefile(name, game:HttpGet(link))
  end
  return getcustomasset(name)
end

local NormalFace = import("https://raw.githubusercontent.com/module-upvr/Others/refs/heads/main/SGNormalFace.png", "SGNormalFace.png")
local KillSFX = import("https://raw.githubusercontent.com/module-upvr/Others/refs/heads/main/WRJingle1.mp3", "WRJingle1.mp3") 
local KillSFX2 = import("https://raw.githubusercontent.com/module-upvr/Others/refs/heads/main/WRJingle2.mp3", "WRJingle2.mp3") 

local function CreatePart(name, parent, weld, offset) 
  local Part = Instance.new("Part")
  Part.Name = name
  Part.Massless = true
  Part.Anchored = false
  Part.CanCollide = false
  Part.CanTouch = false
  Part.CanQuery = false

  if weld then
    local Weld = Instance.new("Weld")
    Weld.C0 = offset or CFrame.new(0, 0, 0) 
    Weld.Part0 = Part
    Weld.Part1 = weld
    Weld.Parent = Part
  end

  Part.Parent = parent
  return Part
end

local function Accs(AccsName, parent, Mesh, Texture, Scale, CF, Welded, Angle) 
  Scale = Scale or Vector3.new(1, 1, 1)
  Angle = Angle or CFrame.Angles(0, 0, 0)
  CF = CF or CFrame.new(0, 0, 0)

  local var_2 = CreatePart(AccsName, parent) 
  var_2.Size = Vector3.new(1, 1, 1)

  local var_3 = Instance.new("SpecialMesh") 
  var_3.Name = "Mesh"
  var_3.MeshId = Mesh
  var_3.TextureId = Texture
  var_3.Scale = Scale
  var_3.Parent = var_2

  local var_4 = Instance.new("Weld") 
  var_4.C0 = CF * Angle
  var_4.Part0 = var_2
  var_4.Part1 = Welded
  var_4.Parent = var_2
  
  return var_2
end

local function Cloth(Id1, Id2, Id3, Parent)
  if Id1 ~= nil then
    local S = Instance.new("Shirt", Parent)
    S.ShirtTemplate = Id1
  end
  if Id2 ~= nil then
    local P = Instance.new("Pants", Parent) 
    P.PantsTemplate = Id2
  end
  if Id3 ~= nil then
    local SG = Instance.new("ShirtGraphic", Parent)
    SG.Graphic = Id3
  end
end

local function D(...)
  for _, v in ipairs({...}) do
    v:Destroy() 
  end
end

local C
local C2
local C3

local function clearC()
  if C then C:Disconnect() C = nil end
  if C2 then C2:Disconnect() C2 = nil end
  if C3 then C3:Disconnect() C3 = nil end
end

local function copyPlaySound(Sound, Parent)
  local clone = Sound:Clone()
  clone.Parent = Parent
  clone:Play()
  clone.Ended:Connect(function() clone:Destroy() end)
  return clone
end

local function DoMorph(P, Name)
  if P.Name == Name and P:GetAttribute("SkinName") == "" then
    for _, v in ipairs({"Torso", "Right Leg", "Left Leg", "Right Arm", "Left Arm"}) do
      local L = P:WaitForChild(v)
      if L then
        L.Transparency = 0
        L.Color = Color3.fromRGB(255,255,255)
      end
    end
    D(P["Slasher Export"], P.Mask, P.Chainsaw.Note, P.Machete.SurfaceAppearance, P.Chainsaw.SurfaceAppearance)

    local hrp = P.HumanoidRootPart

    P.Machete.Transparency = 1
    P.Chainsaw.Color = Color3.new(0, 0, 0) 

    local NewMachete = CreatePart("NewMachete", P.Machete, P.Machete, CFrame.Angles(math.pi/2, 0, 0)) 
    local NMMesh = Instance.new("SpecialMesh", NewMachete) 
    NMMesh.MeshId = "http://www.roblox.com/asset/?id=12221720"
    NewMachete.Color = Color3.new(0, 0, 0) 

    local Kill = Instance.new("Sound", P) 
    Kill.SoundId = KillSFX
    Kill.Name = "WRJingle1"
    Kill.EmitterSize = 12
    Kill.MaxDistance = 60
    Kill.MinDistance = 12
    Kill.Volume = 2
    
    local Kill2 = Instance.new("Sound", P)
    Kill2.SoundId = KillSFX2
    Kill2.Name = "WRJingle2"
    Kill2.EmitterSize = 12
    Kill2.MaxDistance = 60
    Kill2.MinDistance = 12
    Kill2.Volume = 1.5

    for _, v in P:GetChildren() do
      if v:IsA("CharacterMesh") then D(v) end
    end

    local Face = Instance.new("BillboardGui", P.Head) 
    Face.Name = "SGFace"
    Face.Adornee = P.Head
    Face.Size = UDim2.new(1.5, 0, 1.5, 0) 
    Face.StudsOffset = Vector3.new(0, 0, 0.95) 
    Face.AlwaysOnTop = true

    local FaceLabel = Instance.new("ImageLabel", Face) 
    FaceLabel.Name = "FaceIcon"
    FaceLabel.Size = UDim2.new(1, 0, 1, 0) 
    FaceLabel.Image = NormalFace
    FaceLabel.BorderColor3 = Color3.new(0, 0, 0)

    local DisIcon = Instance.new("Decal", P.Torso) 
    DisIcon.Texture = Icon

    C = hrp.ChildAdded:Connect(function(v) 
      if v:IsA("Sound") then
        if v.SoundId == "rbxassetid://80903632607907" then
          copyPlaySound(Kill, hrp)
        elseif v.SoundId == "rbxassetid://106642636433524" then
          copyPlaySound(Kill2, hrp)
        end
      end
    end) 
    
    C2 = Player.CharacterAdded:Once(clearC)
    C3 = P.Humanoid.Died:Once(clearC)
  end
end

Player.CharacterAdded:Connect(function(Rig)
  DoMorph(Rig, "Slasher")
end)
