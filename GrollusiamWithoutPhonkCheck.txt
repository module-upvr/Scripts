local function MakeBG(obj, img)
  local bg = Instance.new("BillboardGui")
  bg.Size = UDim2.new(1, 0, 1, 0)
  bg.StudsOffset = Vector3.new(-3.5, 3.5, 0)
  bg.Adornee = obj
  bg.Brightness = 10
  local image = Instance.new("ImageLabel")
  image.Image = img
  image.Size = UDim2.new(8, 0, 8, 0)
  image.BackgroundTransparency = 1
  image.ImageColor3 = Color3.new(0, 1, 0)
  image.Parent = bg
  
  bg.Parent = obj
end

local function import(link, name)
  local folder = "StickGazerStuffs"
  local fullPath = folder .. "/" .. name
  
  if not isfolder(folder) then
    makefolder(folder)
  end
  
  if not isfile(fullPath) then
    writefile(fullPath, game:HttpGet(link))
  end
  
  local baseName = name:match("([^%.]+)%.%w+$") or name
  local ext = name:match("^.+(%..+)$") or ""
  local tempName = baseName .. "_tempread" .. ext
  
  writefile(tempName, readfile(fullPath))
  local redasset = getcustomasset(tempName)
  
  if isfile(tempName) then
    delfile(tempName)
  end
  
  return redasset
end

local function importinstance(asset, parent)
  local _, imported = pcall(insert.LoadLocalAsset, insert, asset)
  imported.Parent = parent
  return imported
end

local function GetFarthestSurv(c)
  local farthest, minDis = nil, -1
  
  for _, v in Survs:GetChildren() do
    local dis = (c.HumanoidRootPart.Position - v.HumanoidRootPart.Position).Magnitude
    if dis > 0 and dis > minDis then
      minDis = dis
      farthest = v
    end
  end
  return farthest
end

local function GetClosestSurv(c)
  local closest, max = nil, math.huge
  
  for _, v in Survs:GetChildren() do
    local dis = (c.HumanoidRootPart.Position - v.HumanoidRootPart.Position).Magnitude
    if dis < max then
      max = dis
      closest = v
    end
  end
  return closest
end

local function GetClosestNPC(char)
  local nearest, minDistance = nil, math.huge
  for _, desc in workspace:GetDescendants() do
    if desc:IsA("Humanoid") and desc.Parent:IsA("Model") then
      local root = desc.Parent.PrimaryPart or desc.Parent:FindFirstChild("HumanoidRootPart")
      if root and (root.Position - char.HumanoidRootPart.Position).Magnitude < minDistance then
        minDistance = (root.Position - char.HumanoidRootPart.Position).Magnitude
        nearest = desc.Parent
      end
    end
  end
  return nearest
end

      local L = P:WaitForChild(v)
      local L2 = O1R:FindFirstChild(v)

      L.Size = L.Size / 1.121
