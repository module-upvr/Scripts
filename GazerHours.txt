local function Notify(title, text, font, image, duration)

  uiCorner.Parent = frame
  
  local titleLabel = Instance.new("TextLabel")
  titleLabel.Size = UDim2.new(1, -50, 0, 22)
  titleLabel.Position = UDim2.new(0, 10, 0, 5)
  titleLabel.BackgroundTransparency = 1
  titleLabel.Text = title
  titleLabel.TextSize = 16

  textLabel.Size = UDim2.new(1, -20, 1, -32)

local function DED(h, t, o, p1, p2)
  local distance = (p1.Position - p2.Position).Magnitude
  
  local trans = math.clamp(0.5 + ((distance - 50) / 100) * 0.45, 0.5, 0.95)
  local trans2 = math.clamp((distance - 50) / 100, 0, 1)
  
  h.FillTransparency = trans
  t.TextTransparency = trans2
  o.Transparency = trans2
end

local function Esp(obj, Color)
  Color = Color or Color3.new(1, 1, 1)
  
  local C
  local Rig = Player.Character
  
  if not obj.PrimaryPart then
    task.wait(2.5)
  end
  
  local Height = (obj:IsA("Model") and obj:GetExtentsSize() or obj.Size).Y
  local Adornee = obj.PrimaryPart or obj
  local H = obj:FindFirstChildOfClass("Highlight")
  
  if obj.Name == "Alex" then
    local body = obj:WaitForChild("Body")
    for _, v in body:GetChildren() do
      if v:IsA("SurfaceGui") then v.AlwaysOnTop = true end
    end
  elseif obj.Name == "AngryMunci" then
    local renderBody = obj:WaitForChild("Renderbody")
    for _, v in renderBody:GetChildren() do
      if v:IsA("SurfaceGui") then v.AlwaysOnTop = true end
    end
  elseif H then
    H.FillColor = Color
    H.FillTransparency = 0.7
    H.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
  else
    H = Instance.new("Highlight")
    H.Name = "ModifiedESP"
    H.FillColor = Color
    H.FillTransparency = 0.7
    H.OutlineTransparency = 1
    H.Adornee = obj
    H.Parent = obj
  end
  
  local BBG = Instance.new("BillboardGui")
  BBG.Size = UDim2.new(0, 100, 0, 50)
  BBG.AlwaysOnTop = true
  BBG.Adornee = Adornee
  BBG.Active = true
  BBG.StudsOffsetWorldSpace = Vector3.new(0, Height, 0)
  
  local Tag = Instance.new("TextLabel")
  Tag.Text = obj.Name
  Tag.Size = UDim2.new(0, 100, 0, 50)
  Tag.TextSize = 20
  Tag.BackgroundTransparency = 1
  Tag.TextColor3 = Color
  Tag.ZIndex = 10
  Tag.TextYAlignment = Enum.TextYAlignment.Bottom
  Tag.FontFace = TEXT_FONT
  Tag.Parent = BBG
  
  local OL = Instance.new("UIStroke")
  OL.Thickness = 1
  OL.Parent = Tag

  BBG.Parent = Adornee
  
  C = game:GetService("RunService").Heartbeat:Connect(function()
    local hrp = Player.Character and Player.Character.PrimaryPart
    if hrp then DED(H, Tag, OL, hrp, Adornee) end
  end)
  
  obj.Destroying:Once(function()
    C:Disconnect()
    C = nil
  end)
end

local Buttons = {}
local Bools = {}
local buttonCount = 0

local function CreateButton(Text, Name, Parent, Callback)
  Text = Text or "TestButton"
  Name = Name or Text
  Callback = Callback or function() print("TestButton is pressed") end
  
  local column = buttonCount % 3
  local row = math.floor(buttonCount / 3)
  
  local Button = Instance.new("TextButton")
  Button.Position = UDim2.new(column * 0.1, 0, row * 0.1, 0)
  Button.Name = Name
  Button.Text = Text
  Button.Size = UDim2.new(0.1, 0, 0.1, 0)
  Button.TextWrapped = true
  Button.Parent = Parent
  Button.MouseButton1Click:Connect(Callback)
  Buttons[Name] = Button
  buttonCount = buttonCount + 1
end

local function CreateToggleButton(Text, Name, Parent, CallbackTrue, CallbackFalse)
  Text = Text or "TestButton:"
  Name = Name or Text
  CallbackFalse = CallbackFalse or function() print("TestButton is set to False") end
  CallbackTrue = CallbackTrue or function() print("TestButton is set to True") end
  
  Bools[Name] = false
  
  local column = buttonCount % 3
  local row = math.floor(buttonCount / 3)
  
  local Button = Instance.new("TextButton")
  Button.Position = UDim2.new(column * 0.1, 0, row * 0.1, 0)
  Button.Name = Name
  Button.Size = UDim2.new(0.1, 0, 0.1, 0)
  Button.TextWrapped = true
  Button.Text = string.format("%s false", Text)
  Button.Parent = Parent
  Button.MouseButton1Click:Connect(function()
    if Bools[Name] == false then
      CallbackTrue()
      Bools[Name] = true
    elseif Bools[Name] == true then
      CallbackFalse()
      Bools[Name] = false
    end
    Button.Text = string.format("%s %s", Text, tostring(Bools[Name]))
  end)
  Buttons[Name] = Button
  buttonCount = buttonCount + 1
end

local OldSpeedFunc = T.StepMovementSpeed
local NewSpeedFunc = function(p19, p20)
  local Rig = Player.Character
  if not Rig then return end
  local Humanoid = Rig:FindFirstChildOfClass("Humanoid")
  local v21 = p19:GetMovementInput()
  local v22 = v_u_7[v21] or 0.7 
    
  local moveVector = p19.SharedCharacterObject.State.Input.MoveVector
        
  local magnitude = moveVector and moveVector.Magnitude or 0
    
  local v23 = magnitude <= 0.1 and 0.1 or (p19.CurrentSubState == "Climb" and 0.35 or v22)
  local v24 = p20 / 0.016666666666666666
    
  if v24 > 0 and p19.Speed then
    local v25 = v23 - p19.Speed
    local v26
    if v21 == "Sprint" then
      v26 = v25 / 10
    elseif v21 == "Normal" then
      v26 = v25 / 20
    else
      v26 = v25 / 35
    end
    p19.Speed = p19.Speed + v26 * v24
  end
  local SpeedMultiplier = 3.8
  if Humanoid and p19.Speed then
    Humanoid.WalkSpeed = (p19.Speed * 16) * SpeedMultiplier
  end
end
  
local SHOn = function()
  T.StepMovementSpeed = NewSpeedFunc
end
  
local SHOff = function()
  T.StepMovementSpeed = OldSpeedFunc
end
